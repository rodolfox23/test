import { getAxiosInstance } from '@services/axiosInstance';
import type { AppState } from '@storage/interfaces/types';
import { useQuery } from '@tanstack/react-query';

import { queryKeys } from '../reactQuery/queryKeys';
import type { InsuredsBeneficiaresType3Response } from './useRRVVInsuredsBeneficiariesType3.interface';

export interface InsuredsBeneficiariesType3Request {
  appState: AppState;
  userId: string;
  backendId: string;
  lob: string;
  policy: string;
}

interface NanoType3Response {
  Data?: {
    insureds?: unknown[];
    beneficiaries?: unknown[];
  };
  item?: {
    insureds?: unknown[];
    beneficiaries?: unknown[];
  };
  insureds?: unknown[];
  beneficiaries?: unknown[];
}

const normalizeBeneficiaries = (
  beneficiaries: unknown[] | undefined,
): InsuredsBeneficiaresType3Response['beneficiaries'] => {
  if (!beneficiaries || beneficiaries.length === 0) {
    return [];
  }

  return beneficiaries.map((entry, index) => {
    const source = (entry ?? {}) as Record<string, unknown>;

    return {
      type: String(source.type ?? ''),
      name: String(source.name ?? ''),
      beneficiarieRut: String(
        source.beneficiarieRut ?? source.rut ?? source.rutNumber ?? '',
      ),
      relationship: String(source.relationship ?? ''),
      percentageOfAllocation: String(source.percentageOfAllocation ?? ''),
      phone:
        source.phone === undefined ? null : (source.phone as string | null),
      address:
        source.address === undefined ? null : (source.address as string | null),
      email:
        source.email === undefined ? null : (source.email as string | null),
      gender: String(source.gender ?? ''),
      number: Number(source.number ?? index + 1),
      birthdate: String(source.birthdate ?? ''),
      status: String(source.status ?? ''),
      disability: String(source.disability ?? ''),
    };
  });
};

const NANO_ENDPOINT =
  '/v2/tenants/digital/users/products/insuredsBeneficiaries';

export const getRRVVInsuredsBeneficiariesType3 = async ({
  appState,
  userId,
  backendId,
  lob,
  policy,
}: InsuredsBeneficiariesType3Request): Promise<InsuredsBeneficiaresType3Response> => {
  try {
    const { data } = await getAxiosInstance(appState).get<NanoType3Response>(
      NANO_ENDPOINT,
      {
        params: {
          type: 3,
        },
        headers: {
          rut: userId,
          backendId,
          lineOfBusiness: lob,
          policyNumber: policy,
        },
      },
    );

    const insureds =
      data.Data?.insureds ?? data.item?.insureds ?? data.insureds ?? [];
    const rawBeneficiaries =
      data.Data?.beneficiaries ??
      data.item?.beneficiaries ??
      data.beneficiaries ??
      [];

    return {
      insureds,
      beneficiaries: normalizeBeneficiaries(rawBeneficiaries),
    };
  } catch (error: any) {
    if ([404, 400].includes(error.response?.status)) {
      return {
        insureds: [],
        beneficiaries: [],
      };
    }
    
    throw error;
  }
};

export const useRRVVInsuredsBeneficiariesType3 = ({
  appState,
  userId,
  backendId,
  lob,
  policy,
}: InsuredsBeneficiariesType3Request) =>
  useQuery<InsuredsBeneficiaresType3Response, Error>({
    queryKey: [
      queryKeys.rrvvInsuredsBeneficiariesType3,
      userId,
      backendId,
      lob,
      policy,
    ],
    queryFn: () =>
      getRRVVInsuredsBeneficiariesType3({
        appState,
        userId,
        backendId,
        lob,
        policy,
      }),
    retry: false,
  });

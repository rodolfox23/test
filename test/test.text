			(SELECT COUNT(EXCHANGE_RATE) FROM AUA_EXCHANGE_RATE WHERE ID_ACQUIRER = addData.ID_ACQUIRER AND ID_CURRENCY_FROM = openTransaction.MESSAGE_CURRENCY_CODE
				AND FORMAT(VALID_FROM,'MM/yyyy') = FORMAT(openTransaction.TRANSACTION_DATE,'MM/yyyy')) > 0 THEN
				-- Obtengo la ultima tasa de cambio del mes y convierto a colones la transaccion
				(openTransaction.MESSAGE_AMNT * (SELECT EXCHANGE_RATE FROM AUA_EXCHANGE_RATE rate WHERE rate.ID_ACQUIRER = addData.ID_ACQUIRER AND rate.ID_CURRENCY_FROM = openTransaction.MESSAGE_CURRENCY_CODE AND rate.ID_CURRENCY_TO = 188 AND
				rate.VALID_FROM = (SELECT MAX(VALID_FROM) FROM AUA_EXCHANGE_RATE rate2 WHERE rate2.ID_ACQUIRER = addData.ID_ACQUIRER AND rate2.ID_CURRENCY_TO = 188 AND
				rate2.ID_CURRENCY_FROM = openTransaction.MESSAGE_CURRENCY_CODE  AND FORMAT(rate2.VALID_FROM,'MM/yyyy') = FORMAT(openTransaction.TRANSACTION_DATE,'MM/yyyy'))))
				ELSE
				--Si no hay tasas de cambio en el mismo mes de la transaccion Obtengo la ultima tasa de cambio valida
				(openTransaction.MESSAGE_AMNT * (SELECT MAX(EXCHANGE_RATE) FROM AUA_EXCHANGE_RATE rate WHERE
				rate.VALID_FROM = (SELECT MAX(VALID_FROM) FROM AUA_EXCHANGE_RATE rate2 WHERE rate2.ID_ACQUIRER = addData.ID_ACQUIRER AND rate2.ID_CURRENCY_TO = 188 AND
				rate2.ID_CURRENCY_FROM = openTransaction.MESSAGE_CURRENCY_CODE  AND rate2.VALID_FROM <= openTransaction.TRANSACTION_DATE)))
			END
			ELSE openTransaction.MESSAGE_AMNT --Si la transaccion esta en colones no hago conversion
		END MESSAGE_AMOUNT_CO,

	mer.MERCHANT_IDENTIFIER MERCHANT_IDENTIFIER,
	mer.EXTERNAL_MERCHANT_IDENTIFIER,
	mer.FANTASY_NAME MERCHANT_NAME,
	oTrx.CARD_ACCEPTOR_CODE CARD_ACCEPTOR_CODE,
	oTrx.CARD_ACCEPTOR_NAME CARD_ACCEPTOR_NAME,
    mer.ID_MERCHANT ID_MERCHANT,
	br.ID_BRANCH ID_BRANCH,
	oTrx.ID_ORIGINAL_RESPONSE_CODE ORIGINAL_RESPONSE_CODE,
	orgResCode.DESCRIPTION ORIGINAL_RESPONSE_CODE_DESC,
	oTrx.ID_RESPONSE_CODE RESPONSE_CODE,
	resCode.DESCRIPTION RESPONSE_CODE_DESC,
	auMCC.DESCRIPTION MCC_DESCRIPTION,
	oTrx.CARD_ACCEPTOR_TERMINAL_CODE CARD_ACCEPTOR_TERMINAL_CODE,
	prod.ID_BRAND ID_BRAND,
	aTrx.ID_PRODUCT
FROM AU_OPEN_TRANSACTION oTrx
	INNER JOIN AU_TRANSACTION_STATUS stat ON oTrx.ID_TRANSACTION_STATUS = stat.ID_TRANSACTION_STATUS
	LEFT OUTER JOIN AUA_TRX_ACQUIRING_DATA aTrx ON oTrx.ID_TRANSACTION = aTrx.ID_TRANSACTION
	LEFT OUTER JOIN AUA_PRODUCT prod ON prod.ID_PRODUCT = aTrx.ID_PRODUCT
	LEFT OUTER JOIN AUA_BRANCH_DEPARTMENT brDep ON aTrx.ID_BRANCH_DEPARTMENT = brDep.ID_BRANCH_DEPARTMENT
	LEFT OUTER JOIN AUA_BRANCH br ON brDep.ID_BRANCH = br.ID_BRANCH
	LEFT OUTER JOIN AUA_MERCHANT mer ON mer.ID_MERCHANT = br.ID_MERCHANT
	INNER JOIN AUA_ACQUIRER acq ON aTrx.ID_ACQUIRER = acq.ID_ACQUIRER AND acq.ID_ACQUIRER IN (1,2)
    LEFT OUTER JOIN AUA_TRANSACTION_PROFILE trxProf ON trxProf.ID_TRANSACTION_PROFILE = aTrx.ID_TRANSACTION_PROFILE
	LEFT OUTER JOIN AU_RESPONSE_CODE orgResCode ON orgResCode.ID_RESPONSE_CODE = oTrx.ID_ORIGINAL_RESPONSE_CODE
	LEFT OUTER JOIN AU_RESPONSE_CODE resCode ON resCode.ID_RESPONSE_CODE = oTrx.ID_RESPONSE_CODE
	LEFT OUTER JOIN AU_MCC auMCC ON oTrx.MCC_CODE = auMCC.ID_MCC;
